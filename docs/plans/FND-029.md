# FND-029 – Browser tests can expect a page problem

**Phase:** 1 – Foundation
**Depends on:** FND-028 (merged)
**Base branch:** `main`
**Read first:** `CLAUDE.md` §3, `docs/WORKFLOW.md`, `TESTING.md`

## Objective

A browser test can say which page problems it expects, so that the data
status can be tested in the browser with a manifest the server does not have
and with a request that fails. Every other browser test stays as strict as
it is now.

## Design in short

- Today every browser test fails when the page logs an error or a warning,
  throws an uncaught error, has a failed request or receives an HTTP error
  (`frontend/e2e/fixtures.ts`). That stays the rule.
- A test about a failure names what it expects, inside the test, with a new
  fixture: `expectProblem(/pattern/)`. The pattern is matched against the
  recorded text of a problem, for example `HTTP 404: http://…/data/manifest.json`.
- Such a test fails if an expected problem does **not** happen, and still
  fails on any problem it did not name. So an expectation can never hide
  another fault, and it cannot be left behind after the cause has gone.
- The judgement is one small function with no browser in it, `unexplained`,
  so it can be tested directly.
- The rule itself is tested for the first time. Tests that must fail under
  the rule are marked with Playwright's `test.fail()`, which passes a test
  only when it fails; they are reported as passed, never as skipped.
- The expectation is a function called in the test, not a `test.use` option:
  Playwright reads a two-element list given to `test.use` as a value with
  settings, so a list of two patterns breaks (the planner tried it).
- The fixture `pageProblems` is renamed `pageHealth`, because it now holds
  the expectations as well. No test names it.

### 1. `frontend/e2e/fixtures.ts` – the whole file

```ts
import { expect, test as base } from '@playwright/test'

/**
 * What is wrong with the problems a page reported, given the problems the
 * test said it expects. Every problem must be an expected one, and every
 * expected problem must have happened. An empty list means all is well.
 */
export function unexplained(problems: readonly string[], expected: readonly RegExp[]): string[] {
  const unexpected = problems.filter(
    (problem) => !expected.some((pattern) => pattern.test(problem)),
  )
  const missing = expected
    .filter((pattern) => !problems.some((problem) => pattern.test(problem)))
    .map((pattern) => `expected, but did not happen: ${String(pattern)}`)
  return [...unexpected, ...missing]
}

type Fixtures = {
  /**
   * Says that this test expects the page to report a problem whose recorded
   * text matches the pattern. Do not give the pattern the `g` or `y` flag:
   * such a pattern remembers where it stopped.
   */
  expectProblem: (pattern: RegExp) => void
  pageHealth: { readonly problems: string[]; readonly expected: RegExp[] }
}

/**
 * Every browser test uses this `test`. It records anything that shows the page
 * is unhealthy and fails the test if there was any, even when the test's own
 * assertions passed. A test about a failure names the problems it expects with
 * `expectProblem`; it then fails if one of them does not happen, and still
 * fails on any other.
 */
export const test = base.extend<Fixtures>({
  pageHealth: [
    async ({ page }, provide) => {
      const problems: string[] = []
      const expected: RegExp[] = []
      page.on('console', (message) => {
        const type = message.type()
        if (type === 'error' || type === 'warning') {
          problems.push(`console ${type}: ${message.text()}`)
        }
      })
      page.on('pageerror', (error) => {
        problems.push(`uncaught error: ${error.message}`)
      })
      page.on('requestfailed', (request) => {
        problems.push(`request failed: ${request.method()} ${request.url()}`)
      })
      page.on('response', (response) => {
        if (response.status() >= 400) {
          problems.push(`HTTP ${response.status()}: ${response.url()}`)
        }
      })
      await provide({ problems, expected })
      expect(unexplained(problems, expected), 'the page reported problems').toEqual([])
    },
    { auto: true },
  ],
  expectProblem: async ({ pageHealth }, provide) => {
    await provide((pattern) => {
      pageHealth.expected.push(pattern)
    })
  },
})

export { expect }
```

### 2. `frontend/e2e/fixtures.spec.ts` – 16 browser tests

```ts
import type { Page } from '@playwright/test'
import { expect, test, unexplained } from './fixtures'

/**
 * The rule every browser test runs under: a page that reports a problem fails
 * its test, unless the test said it expects exactly that problem.
 */

/** Makes the page report an uncaught error with the given message. */
async function breakPage(page: Page, message: string) {
  const reported = page.waitForEvent('pageerror')
  await page.evaluate((text) => {
    setTimeout(() => {
      throw new Error(text)
    })
  }, message)
  await reported
}

/** Makes the page write a line to its console. */
async function writeToConsole(page: Page, method: 'error' | 'warn' | 'info', text: string) {
  await page.evaluate(`console.${method}(${JSON.stringify(text)})`)
}

test.describe('judging the problems a page reported', () => {
  test('finds nothing wrong when there are no problems and none is expected', () => {
    expect(unexplained([], [])).toEqual([])
  })

  test('names every problem nobody expected', () => {
    expect(unexplained(['console error: boom', 'HTTP 500: /x'], [])).toEqual([
      'console error: boom',
      'HTTP 500: /x',
    ])
  })

  test('accepts a problem the test expects', () => {
    expect(unexplained(['HTTP 404: /data/manifest.json'], [/^HTTP 404: /])).toEqual([])
  })

  test('accepts an expected problem that happens more than once', () => {
    expect(unexplained(['HTTP 404: /a', 'HTTP 404: /a'], [/^HTTP 404: \/a$/])).toEqual([])
  })

  test('names an expected problem that did not happen', () => {
    expect(unexplained([], [/^HTTP 404: /])).toEqual([
      'expected, but did not happen: /^HTTP 404: /',
    ])
  })

  test('names another problem beside the expected one', () => {
    expect(unexplained(['HTTP 404: /a', 'console error: boom'], [/^HTTP 404: /])).toEqual([
      'console error: boom',
    ])
  })

  test('wants every expected problem, not just one of them', () => {
    expect(unexplained(['HTTP 404: /a'], [/^HTTP 404: /, /^console error: /])).toEqual([
      'expected, but did not happen: /^console error: /',
    ])
  })
})

// `test.fail()` turns a test round: it passes only when the test fails. These
// tests do nothing wrong themselves, so what fails them is the rule.
test.describe('a test that expects no problem', () => {
  test('passes on a healthy page', async ({ page }) => {
    await page.goto('/')
  })

  test('passes when the page only writes information to its console', async ({ page }) => {
    await page.goto('/')

    await writeToConsole(page, 'info', 'all is well')
  })

  test('fails when the page reports one', async ({ page }) => {
    test.fail()
    await page.goto('/')

    await breakPage(page, 'boom')
  })
})

test.describe('a test that expects a problem', () => {
  test('passes when the page reports it', async ({ page, expectProblem }) => {
    expectProblem(/^uncaught error: boom$/)
    await page.goto('/')

    await breakPage(page, 'boom')
  })

  test('fails when the page does not report it', async ({ page, expectProblem }) => {
    test.fail()
    expectProblem(/^uncaught error: boom$/)
    await page.goto('/')
  })

  test('fails when the page reports another problem as well', async ({ page, expectProblem }) => {
    test.fail()
    expectProblem(/^uncaught error: boom$/)
    await page.goto('/')

    await breakPage(page, 'boom')
    await breakPage(page, 'bang')
  })

  test('counts an error in the console as a problem', async ({ page, expectProblem }) => {
    expectProblem(/^console error: boom$/)
    await page.goto('/')

    await writeToConsole(page, 'error', 'boom')
  })

  test('counts a warning in the console as a problem', async ({ page, expectProblem }) => {
    expectProblem(/^console warning: careful$/)
    await page.goto('/')

    await writeToConsole(page, 'warn', 'careful')
  })

  test('passes when the page reports each of two expected problems', async ({
    page,
    expectProblem,
  }) => {
    expectProblem(/^uncaught error: boom$/)
    expectProblem(/^uncaught error: bang$/)
    await page.goto('/')

    await breakPage(page, 'boom')
    await breakPage(page, 'bang')
  })
})
```

### 3. `frontend/e2e/status.spec.ts` – two tests added

Add these two tests at the end of `test.describe('data status', …)`, after
"keeps the rest of the page working when the data is unavailable". Nothing
else in the file changes.

```ts
  // A missing file and a failed request are page problems, which fail every
  // browser test (see `fixtures.ts`). These two tests say that they expect them.
  test('says the data is unavailable when the server does not have the manifest', async ({
    page,
    expectProblem,
  }) => {
    expectProblem(/^HTTP 404: http:\/\/[^/]+\/data\/manifest\.json$/)
    expectProblem(/^console error: Failed to load resource: .* status of 404 /)
    await page.route(MANIFEST, (route) =>
      route.fulfill({ status: 404, contentType: 'text/html', body: '<h1>Not found</h1>' }),
    )
    await page.goto('/')
    await expect(dataStatus(page)).toHaveText('Data unavailable')

    await page.getByRole('button', { name: 'Events' }).click()

    await expect(page.getByRole('button', { name: 'Events' })).toHaveAttribute(
      'aria-current',
      'page',
    )
  })

  test('says the data is unavailable when the request for the manifest fails', async ({
    page,
    expectProblem,
  }) => {
    expectProblem(/^request failed: GET http:\/\/[^/]+\/data\/manifest\.json$/)
    expectProblem(/^console error: Failed to load resource: net::ERR_FAILED$/)
    await page.route(MANIFEST, (route) => route.abort())
    await page.goto('/')
    await expect(dataStatus(page)).toHaveText('Data unavailable')

    await page.getByRole('button', { name: 'Events' }).click()

    await expect(page.getByRole('button', { name: 'Events' })).toHaveAttribute(
      'aria-current',
      'page',
    )
  })
```

## Acceptance criteria

1. A browser test that names no problem behaves exactly as before: it fails
   on a console error or warning, an uncaught error, a failed request or an
   HTTP error, and passes on a healthy page.
2. A test that names its expected problems passes only when every one of
   them happens and nothing else does.
3. The data status says `Data unavailable` when the manifest is answered
   with HTTP 404 and when the request fails, and the rest of the page keeps
   working.
4. `npm run e2e` reports 80 passed, with no test skipped; `npm test` still
   reports 199 passed in 12 files; every other gate passes.
5. Only the files under "Scope" changed; no dependency changed.
