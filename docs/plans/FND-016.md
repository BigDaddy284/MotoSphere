# FND-016 – Manifest reader in the frontend

**Phase:** 1 – Foundation
**Depends on:** FND-021 (merged)
**Base branch:** `main`
**Read first:** `CLAUDE.md` §3, §11, §14, `docs/WORKFLOW.md`, `ARCHITECTURE.md` §3 and §6

## Objective

The frontend can fetch the data manifest that the pipeline publishes and
check it, and reports every way that can go wrong as a plain result instead
of an error. This is the reading half of "what the pipeline publishes
reaches the user's screen"; nothing is shown on the page yet.

## Design in short

- One module, `src/data/manifest.ts`, with no dependency and no React.
- `parseManifest(value)` checks a parsed JSON value field by field and
  returns either the manifest or the reason there is none. Published data is
  treated as untrusted: nothing is used before it has been checked.
- The format version is checked first. A manifest in another version of the
  format is `unsupported` whatever else it holds, so that an old copy of the
  application meeting newer data can say so. Unknown extra fields are
  ignored, because a field may be added without raising the format version.
- `loadManifest(baseUrl, options)` fetches `data/manifest.json` under the
  application's base path and never throws. Its result is one of four
  things: the manifest, `unavailable` (the request failed or the server
  answered with an error), `malformed` (the answer is not a manifest) or
  `unsupported`.
- The field names in the file are the pipeline's (`format_version`); the
  manifest the application works with uses the frontend's naming
  (`formatVersion`). The counts become a `ReadonlyMap`, so that a record kind
  with a name like `constructor` can never collide with an object's own
  members.
- The fetch function can be passed in, so the unit tests need no network and
  no global stub.

### 1. `frontend/src/data/manifest.ts`

```ts
/**
 * The manifest is the first published data file the application reads. It
 * says which version of the data format the files use, when they were
 * generated and how many records of each kind they hold.
 *
 * The file is written by the data pipeline (`python -m motosphere export`).
 * Nothing in it is trusted: every field is checked before it is used.
 */

/** The version of the published data format this application can read. */
export const SUPPORTED_FORMAT_VERSION = 1

export type Manifest = {
  readonly formatVersion: number
  /** When the data was generated: a moment in UTC, to the second, as `YYYY-MM-DDTHH:MM:SSZ`. */
  readonly generatedAt: string
  /** The MotoSphere version that generated the data. */
  readonly applicationVersion: string
  /** The number of published records of each kind, for example `tracks`. */
  readonly counts: ReadonlyMap<string, number>
}

/**
 * Why there is no manifest to show:
 * - `unavailable`: the file could not be fetched.
 * - `malformed`: the file is not a manifest.
 * - `unsupported`: the file is a manifest in a data format this application
 *   cannot read, usually because the application is older than the data.
 */
export type ManifestProblem = 'unavailable' | 'malformed' | 'unsupported'

export type ManifestResult =
  | { readonly ok: true; readonly manifest: Manifest }
  | { readonly ok: false; readonly problem: ManifestProblem }

const MOMENT_LENGTH = 'YYYY-MM-DDTHH:MM:SSZ'.length
const RECORD_KIND = /^[a-z][a-z_]*$/

function failed(problem: ManifestProblem): ManifestResult {
  return { ok: false, problem }
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

function isCount(value: unknown): value is number {
  return typeof value === 'number' && Number.isSafeInteger(value) && value >= 0
}

/** True for a real moment written exactly as `YYYY-MM-DDTHH:MM:SSZ`. */
function isMoment(value: unknown): value is string {
  if (typeof value !== 'string' || value.length !== MOMENT_LENGTH) {
    return false
  }
  const time = Date.parse(value)
  // Writing the moment out again and comparing accepts this one spelling only,
  // and rejects dates that do not exist, such as 30 February.
  return !Number.isNaN(time) && new Date(time).toISOString() === value.replace('Z', '.000Z')
}

function readCounts(value: unknown): ReadonlyMap<string, number> | undefined {
  if (!isRecord(value)) {
    return undefined
  }
  const counts = new Map<string, number>()
  for (const [kind, count] of Object.entries(value)) {
    if (!RECORD_KIND.test(kind) || !isCount(count)) {
      return undefined
    }
    counts.set(kind, count)
  }
  return counts
}

/**
 * Checks a parsed JSON value and returns the manifest it holds.
 *
 * The format version is checked first: a manifest in another version of the
 * format is `unsupported` whatever else it contains. Fields this application
 * does not know are ignored.
 */
export function parseManifest(value: unknown): ManifestResult {
  if (!isRecord(value)) {
    return failed('malformed')
  }
  const formatVersion = value['format_version']
  if (
    typeof formatVersion !== 'number' ||
    !Number.isSafeInteger(formatVersion) ||
    formatVersion < 1
  ) {
    return failed('malformed')
  }
  if (formatVersion !== SUPPORTED_FORMAT_VERSION) {
    return failed('unsupported')
  }

  const generatedAt = value['generated_at']
  const applicationVersion = value['application_version']
  const counts = readCounts(value['counts'])
  if (
    !isMoment(generatedAt) ||
    typeof applicationVersion !== 'string' ||
    applicationVersion === '' ||
    counts === undefined
  ) {
    return failed('malformed')
  }
  return { ok: true, manifest: { formatVersion, generatedAt, applicationVersion, counts } }
}

/**
 * The address of the manifest for an application served from `baseUrl`
 * (Vite's `import.meta.env.BASE_URL`, for example `/` or `/MotoSphere/`).
 */
export function manifestUrl(baseUrl: string): string {
  return `${baseUrl.endsWith('/') ? baseUrl : `${baseUrl}/`}data/manifest.json`
}

export type LoadManifestOptions = {
  /** Stops the request, for example when the part of the page that asked for it goes away. */
  readonly signal?: AbortSignal
  /** The function that fetches; the browser's `fetch` unless a test passes its own. */
  readonly fetcher?: typeof fetch
}

/**
 * Fetches and checks the manifest. Never throws: every failure is a result
 * with the reason.
 */
export async function loadManifest(
  baseUrl: string,
  { signal, fetcher = fetch }: LoadManifestOptions = {},
): Promise<ManifestResult> {
  // `no-cache` makes the browser ask the server whether its copy is still
  // current, so newly published data is seen without a forced reload.
  const init: RequestInit =
    signal === undefined ? { cache: 'no-cache' } : { cache: 'no-cache', signal }
  let response: Response
  try {
    response = await fetcher(manifestUrl(baseUrl), init)
  } catch {
    return failed('unavailable')
  }
  if (!response.ok) {
    return failed('unavailable')
  }
  let content: unknown
  try {
    content = await response.json()
  } catch {
    return failed('malformed')
  }
  return parseManifest(content)
}
```

Notes:

- `exactOptionalPropertyTypes` is on, so `signal: undefined` may not be
  passed to `fetch`; hence the two forms of `init`.
- In `isMoment` every condition is needed: the length rejects years of more
  than four digits, the `NaN` check keeps `toISOString` from throwing, and
  writing the moment out again rejects every other spelling and every date
  that does not exist.

### 2. `frontend/src/data/manifest.test.ts` – 62 cases

```ts
import { describe, expect, it, vi } from 'vitest'
import {
  loadManifest,
  manifestUrl,
  parseManifest,
  SUPPORTED_FORMAT_VERSION,
  type ManifestResult,
} from './manifest.ts'

// Exactly what `python -m motosphere export` wrote on 17 November 2026.
const EXPORTED = `{
  "application_version": "0.1.0",
  "counts": {},
  "format_version": 1,
  "generated_at": "2026-11-17T12:53:59Z"
}
`

function valid(changes: Record<string, unknown> = {}): Record<string, unknown> {
  return {
    format_version: 1,
    generated_at: '2026-11-17T12:53:59Z',
    application_version: '0.1.0',
    counts: { tracks: 3, events: 12 },
    ...changes,
  }
}

function without(field: string): Record<string, unknown> {
  return Object.fromEntries(Object.entries(valid()).filter(([name]) => name !== field))
}

const MALFORMED: ManifestResult = { ok: false, problem: 'malformed' }
const UNAVAILABLE: ManifestResult = { ok: false, problem: 'unavailable' }
const UNSUPPORTED: ManifestResult = { ok: false, problem: 'unsupported' }

function jsonResponse(content: unknown, status = 200): Response {
  return new Response(JSON.stringify(content), {
    status,
    headers: { 'Content-Type': 'application/json' },
  })
}

function fetcherGiving(response: Response) {
  return vi.fn<typeof fetch>().mockResolvedValue(response)
}

describe('parseManifest', () => {
  it('reads the manifest the export command writes', () => {
    const result = parseManifest(JSON.parse(EXPORTED))

    expect(result).toEqual({
      ok: true,
      manifest: {
        formatVersion: 1,
        generatedAt: '2026-11-17T12:53:59Z',
        applicationVersion: '0.1.0',
        counts: new Map(),
      },
    })
  })

  it('reads the number of records of each kind', () => {
    const result = parseManifest(valid())

    expect(result.ok && [...result.manifest.counts]).toEqual([
      ['tracks', 3],
      ['events', 12],
    ])
  })

  it('reads the format version this application supports', () => {
    expect(SUPPORTED_FORMAT_VERSION).toBe(1)
    expect(parseManifest(valid({ format_version: SUPPORTED_FORMAT_VERSION })).ok).toBe(true)
  })

  it('ignores fields it does not know', () => {
    const result = parseManifest(valid({ files: ['tracks.json'] }))

    expect(result.ok && Object.keys(result.manifest).toSorted()).toEqual([
      'applicationVersion',
      'counts',
      'formatVersion',
      'generatedAt',
    ])
  })

  it.each([
    ['nothing', null],
    ['a list', [valid()]],
    ['a text', EXPORTED],
    ['a number', 1],
    ['a yes or no', true],
  ])('rejects %s in place of a manifest', (_name, value) => {
    expect(parseManifest(value)).toEqual(MALFORMED)
  })

  it.each(['format_version', 'generated_at', 'application_version', 'counts'])(
    'rejects a manifest without %s',
    (field) => {
      expect(parseManifest(without(field))).toEqual(MALFORMED)
    },
  )

  it.each([
    ['zero', 0],
    ['a negative number', -1],
    ['a fraction', 1.5],
    ['a text', '1'],
    ['nothing', null],
  ])('rejects a format version that is %s', (_name, value) => {
    expect(parseManifest(valid({ format_version: value }))).toEqual(MALFORMED)
  })

  it.each([
    ['with a time zone offset', '2026-11-17T12:53:59+00:00'],
    ['with fractions of a second', '2026-11-17T12:53:59.000Z'],
    ['without the T', '2026-11-17 12:53:59Z'],
    ['without a time', '2026-11-17'],
    ['on a day that does not exist', '2026-02-30T00:00:00Z'],
    ['in a month that does not exist', '2026-13-01T00:00:00Z'],
    ['at an hour that does not exist', '2026-11-17T24:00:00Z'],
    ['in a year of more than four digits', '+010000-01-01T00:00:00Z'],
    ['in small letters', '2026-11-17t12:53:59z'],
    ['in words', 'yesterday'],
    ['that is empty', ''],
    ['as a number', 1_763_384_039],
    ['with something after it', '2026-11-17T12:53:59Z\n<script>'],
  ])('rejects a generation time %s', (_name, value) => {
    expect(parseManifest(valid({ generated_at: value }))).toEqual(MALFORMED)
  })

  it.each([
    ['empty', ''],
    ['a number', 5],
    ['nothing', null],
  ])('rejects an application version that is %s', (_name, value) => {
    expect(parseManifest(valid({ application_version: value }))).toEqual(MALFORMED)
  })

  it.each([
    ['nothing', null],
    ['a list', [3]],
    ['an empty list', []],
    ['a text', 'none'],
    ['a negative number', { tracks: -1 }],
    ['a fraction', { tracks: 1.5 }],
    ['a number written as text', { tracks: '3' }],
    ['a number too large to be exact', { tracks: 2 ** 53 }],
    ['a kind with a capital letter', { Tracks: 1 }],
    ['a kind with no name', { '': 1 }],
    ['a kind with spaces', { 'tracks per country': 1 }],
    ['a kind that starts with an underscore', JSON.parse('{"__proto__": 1}')],
  ])('rejects counts holding %s', (_name, value) => {
    expect(parseManifest(valid({ counts: value }))).toEqual(MALFORMED)
  })

  it('says a newer data format is unsupported', () => {
    expect(parseManifest(valid({ format_version: 2 }))).toEqual(UNSUPPORTED)
  })

  it('checks the format version before anything else', () => {
    expect(parseManifest({ format_version: 2, layout: 'something new' })).toEqual(UNSUPPORTED)
  })
})

describe('manifestUrl', () => {
  it.each([
    ['/', '/data/manifest.json'],
    ['/MotoSphere/', '/MotoSphere/data/manifest.json'],
    ['/MotoSphere', '/MotoSphere/data/manifest.json'],
  ])('for the base path %s is %s', (baseUrl, expected) => {
    expect(manifestUrl(baseUrl)).toBe(expected)
  })
})

describe('loadManifest', () => {
  it('fetches the manifest under the base path, asking the server for the current copy', async () => {
    const fetcher = fetcherGiving(jsonResponse(valid()))

    await loadManifest('/MotoSphere/', { fetcher })

    expect(fetcher).toHaveBeenCalledTimes(1)
    expect(fetcher).toHaveBeenCalledWith('/MotoSphere/data/manifest.json', { cache: 'no-cache' })
  })

  it('returns the manifest it fetched', async () => {
    const fetcher = fetcherGiving(new Response(EXPORTED))

    const result = await loadManifest('/', { fetcher })

    expect(result.ok && result.manifest.generatedAt).toBe('2026-11-17T12:53:59Z')
  })

  it('passes on the signal that stops the request', async () => {
    const fetcher = fetcherGiving(jsonResponse(valid()))
    const { signal } = new AbortController()

    await loadManifest('/', { fetcher, signal })

    expect(fetcher).toHaveBeenCalledWith('/data/manifest.json', { cache: 'no-cache', signal })
  })

  it("uses the browser's fetch unless it is given another", async () => {
    const browserFetch = vi.spyOn(globalThis, 'fetch').mockResolvedValue(jsonResponse(valid()))

    const result = await loadManifest('/')

    expect(result.ok).toBe(true)
    expect(browserFetch).toHaveBeenCalledWith('/data/manifest.json', { cache: 'no-cache' })
  })

  it('is unavailable when the request fails', async () => {
    const fetcher = vi.fn<typeof fetch>().mockRejectedValue(new TypeError('Failed to fetch'))

    await expect(loadManifest('/', { fetcher })).resolves.toEqual(UNAVAILABLE)
  })

  it('is unavailable when the request is stopped', async () => {
    const stopped = new DOMException('The operation was aborted.', 'AbortError')
    const fetcher = vi.fn<typeof fetch>().mockRejectedValue(stopped)

    await expect(loadManifest('/', { fetcher })).resolves.toEqual(UNAVAILABLE)
  })

  it.each([404, 500])(
    'is unavailable when the server answers %d, whatever it sends',
    async (status) => {
      const fetcher = fetcherGiving(jsonResponse(valid(), status))

      await expect(loadManifest('/', { fetcher })).resolves.toEqual(UNAVAILABLE)
    },
  )

  it('is malformed when the answer is a web page instead of data', async () => {
    const page = new Response('<!doctype html><title>MotoSphere</title>', {
      headers: { 'Content-Type': 'text/html' },
    })

    await expect(loadManifest('/', { fetcher: fetcherGiving(page) })).resolves.toEqual(MALFORMED)
  })

  it('is malformed when the data is not a manifest', async () => {
    const fetcher = fetcherGiving(jsonResponse({ tracks: [] }))

    await expect(loadManifest('/', { fetcher })).resolves.toEqual(MALFORMED)
  })

  it('is unsupported when the data is in a newer format', async () => {
    const fetcher = fetcherGiving(jsonResponse(valid({ format_version: 2 })))

    await expect(loadManifest('/', { fetcher })).resolves.toEqual(UNSUPPORTED)
  })
})
```

## Acceptance criteria

1. `parseManifest` returns the manifest for the text the export command
   writes, with the counts as a map.
2. A value that is not an object, a missing field, or a field with a wrong
   type or an impossible value gives `malformed`. That includes a generation
   time in any spelling but `YYYY-MM-DDTHH:MM:SSZ` or on a date that does not
   exist, a count that is negative, fractional or too large to be exact, and
   a record kind that is not lower-case letters and underscores.
3. A format version other than the supported one gives `unsupported`, even
   when nothing else in the file can be read.
4. `loadManifest` asks for `data/manifest.json` under the base path with
   `cache: 'no-cache'`, passes a given signal on, and never throws: a failed
   or stopped request and an error answer give `unavailable`; an answer that
   is not JSON or not a manifest gives `malformed`.
5. `npm test` reports 177 passed in 9 files; `npm run e2e` still reports 47
   passed; every other gate passes.
6. Only the files under "Scope" changed; no dependency changed.
